cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(acosh_validation LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
add_executable(validate validate.asc)
target_compile_options(validate PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_link_libraries(validate PRIVATE tiling_api register platform unified_dlog dl m graph_base)
foreach(variant baseline tile4096)
    add_executable(${variant} validate.asc)
    target_compile_definitions(${variant} PRIVATE VALIDATION_KERNEL="kernel_${variant}.asc")
    target_compile_options(${variant} PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
    target_link_libraries(${variant} PRIVATE tiling_api register platform unified_dlog dl m graph_base)
endforeach()
